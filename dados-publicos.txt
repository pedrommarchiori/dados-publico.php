        // Função para Atualizar Cidades com Base no Estado
        function atualizarCidades(estadoSelectId, cidadeSelectId) {
            const estado = document.getElementById(estadoSelectId).value;
            const cidadeSelect = document.getElementById(cidadeSelectId);
            cidadeSelect.innerHTML = '<option value="">Todos</option>';

            if (estado) {
                const cidades = [...new Set(dadosOriginais
                    .filter(d => d[estadoSelectId.replace('filtro_', '')] === estado)
                    .map(d => d[cidadeSelectId.replace('filtro_', '')]))
                ].sort();
                cidades.forEach(cidade => {
                    const option = document.createElement('option');
                    option.value = cidade;
                    option.textContent = cidade;
                    cidadeSelect.appendChild(option);
                });
            }
        }

        // Inicializar Cidades
        document.getElementById('filtro_estado_1opcao').addEventListener('change', () => atualizarCidades('filtro_estado_1opcao', 'filtro_cidade_1opcao'));
        document.getElementById('filtro_estado_2opcao').addEventListener('change', () => atualizarCidades('filtro_estado_2opcao', 'filtro_cidade_2opcao'));

        // Função de Filtro
        function mpf_filtrar(opcao) {
            const prefix = opcao === '1opcao' ? '1opcao' : '2opcao';
            const estado = document.getElementById(`filtro_estado_${prefix}`).value;
            const cidade = document.getElementById(`filtro_cidade_${prefix}`).value;
            const pontuacao = document.getElementById(`filtro_pontuacao_${prefix}`).value;
            const cotas = document.getElementById(`filtro_cotas_${prefix}`).value;

            const cards = document.querySelectorAll(`#dados_${prefix} .mpf-card`);
            cards.forEach(card => {
                const matchEstado = estado === "" || card.dataset[`estado${prefix}`] === estado;
                const matchCidade = cidade === "" || card.dataset[`cidade${prefix}`] === cidade;
                const matchPontuacao = pontuacao === "" || card.dataset.pontuacao === pontuacao;
                const matchCotas = cotas === "" || card.dataset.cotas === cotas;

                card.style.display = matchEstado && matchCidade && matchPontuacao && matchCotas ? "block" : "none";
            });

            atualizarGraficos(opcao);
        }

        // Toggle Chart
        function toggleChart(button) {
            const content = button.parentElement.querySelector('.chart-content');
            content.classList.toggle('minimized');
            button.textContent = content.classList.contains('minimized') ? '+' : '−';
        }

        // Toggle Card
        function toggleCard(button) {
            const content = button.parentElement.querySelector('.card-content');
            content.classList.toggle('minimized');
            button.textContent = content.classList.contains('minimized') ? '+' : '−';
        }

        // Tab Navigation
        function openTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(content => {
                content.classList.remove('active');
            });
            document.querySelectorAll('.tab-button').forEach(button => {
                button.classList.remove('active');
            });

            document.getElementById(tabId).classList.add('active');
            document.querySelector(`[onclick="openTab('${tabId}')"]`).classList.add('active');
        }

        // Atualizar Gráficos com Filtros
        function atualizarGraficos(opcao) {
            const prefix = opcao === '1opcao' ? '1opcao' : '2opcao';
            const cardsVisiveis = Array.from(document.querySelectorAll(`#dados_${prefix} .mpf-card`)).filter(card => card.style.display !== "none");
            const novosDados = {
                pontuacao: {},
                cotas: {},
                estado: {}
            };

            cardsVisiveis.forEach(card => {
                const pontuacao = card.dataset.pontuacao;
                const cotas = card.dataset.cotas;
                const estado = card.dataset[`estado${prefix}`];

                novosDados.pontuacao[pontuacao] = (novosDados.pontuacao[pontuacao] || 0) + 1;
                novosDados.cotas[cotas] = (novosDados.cotas[cotas] || 0) + 1;
                novosDados.estado[estado] = (novosDados.estado[estado] || 0) + 1;
            });

            const chartPontuacao = opcao === '1opcao' ? chartPontuacao1 : chartPontuacao2;
            const chartCotas = opcao === '1opcao' ? chartCotas1 : chartCotas2;
            const chartEstado = opcao === '1opcao' ? chartEstado1 : chartEstado2;

            chartPontuacao.data.labels = Object.keys(novosDados.pontuacao);
            chartPontuacao.data.datasets[0].data = Object.values(novosDados.pontuacao);
            chartPontuacao.update();

            chartCotas.data.labels = Object.keys(novosDados.cotas);
            chartCotas.data.datasets[0].data = Object.values(novosDados.cotas);
            chartCotas.update();

            chartEstado.data.labels = Object.keys(novosDados.estado);
            chartEstado.data.datasets[0].data = Object.values(novosDados.estado);
            chartEstado.update();
        }
